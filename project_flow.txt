```

       ⬇                                 (sends files/text)
Frontend returns video <- Backend calls -> Colab (Flask + Wav2Lip) (GPU)
```

```python
# Run this cell (it will block and print a public ngrok URL)
from flask import Flask, request, send_file
from flask_ngrok import run_with_ngrok
import subprocess, uuid, os
app = Flask(__name__)
run_with_ngrok(app)  # prints public URL once server is running

WAV2LIP_DIR = "/content/Wav2Lip"
CHECKPOINT_PATH = os.path.join(WAV2LIP_DIR, "checkpoints", "wav2lip_gan.pth")  # adjust if different

@app.route("/run-wav2lip", methods=["POST"])
def run_wav2lip():
    # expects multipart with keys: image (file) and audio (file)
    uid = uuid.uuid4().hex
    img_path = f"/content/in_{uid}.jpg"
    aud_path = f"/content/in_{uid}.wav"
    out_path = f"/content/out_{uid}.mp4"

    if "image" not in request.files or "audio" not in request.files:
        return {"error": "Provide image and audio files"}, 400

    request.files["image"].save(img_path)
    request.files["audio"].save(aud_path)

    # Run the Wav2Lip inference command (adjust args if repo uses a different script/flag)
    cmd = f"python {WAV2LIP_DIR}/inference.py --checkpoint_path {CHECKPOINT_PATH} --face {img_path} --audio {aud_path} --outfile {out_path}"
    print("Running:", cmd)
    try:
        subprocess.check_call(cmd, shell=True)
    except subprocess.CalledProcessError as e:
        print("Wav2Lip failed:", e)
        return {"error": "Wav2Lip inference failed"}, 500

    # Return generated video
    return send_file(out_path, mimetype="video/mp4")

if __name__ == "__main__":
    app.run()
```

Create `backend/app/colab_client.py` — handles calling Colab:

```python
import requests

# Replace with the ngrok URL printed from Colab when running
COLAB_URL = "https://YOUR_COLAB_NGROK_URL/run-wav2lip"

def call_colab(image_path: str, audio_path: str, timeout=300):
    files = {
        "image": open(image_path, "rb"),
        "audio": open(audio_path, "rb")
    }
    resp = requests.post(COLAB_URL, files=files, timeout=timeout)
    resp.raise_for_status()
    out_path = image_path + ".result.mp4"
    with open(out_path, "wb") as f:
        f.write(resp.content)
    return out_path
```

Create `backend/app/tts.py` — text→audio helper:

```python
from gtts import gTTS
from pydub import AudioSegment
import os

def text_to_wav(text: str, out_wav_path: str):
    # Save as temporary mp3 then convert to wav at 16kHz mono
    tmp_mp3 = out_wav_path + ".mp3"
    tts = gTTS(text)
    tts.save(tmp_mp3)

    # Convert to wav 16000Hz mono (Wav2Lip often expects standard rates)
    audio = AudioSegment.from_mp3(tmp_mp3)
    audio = audio.set_frame_rate(16000).set_channels(1)
    audio.export(out_wav_path, format="wav")
    os.remove(tmp_mp3)
    return out_wav_path
```

> gTTS uses the internet. If you want offline TTS, switch to pyttsx3 or Coqui TTS.

Create `backend/app/utils.py` for safe file saving:

```python
import uuid
from pathlib import Path

TMP_DIR = Path("tmp")
TMP_DIR.mkdir(exist_ok=True)

def save_upload_file(upload_file, prefix="file"):
    uid = uuid.uuid4().hex
    ext = Path(upload_file.filename).suffix or ""
    path = TMP_DIR / f"{prefix}_{uid}{ext}"
    with open(path, "wb") as f:
        f.write(upload_file.file.read())
    return str(path)
```

Create `backend/app/main.py` — FastAPI app:

```python
from fastapi import FastAPI, File, UploadFile, Form
from fastapi.responses import FileResponse
from .utils import save_upload_file
from .tts import text_to_wav
from .colab_client import call_colab
from pathlib import Path

app = FastAPI()

@app.post("/generate-from-text")
async def generate_from_text(image: UploadFile = File(...), text: str = Form(...)):
    # Save uploaded image
    image_path = save_upload_file(image, prefix="img")
    # Create TTS audio
    audio_path = str(Path(image_path).with_suffix(".wav"))
    text_to_wav(text, audio_path)
    # Call Colab to run Wav2Lip
    out_video = call_colab(image_path, audio_path)
    return FileResponse(out_video, media_type="video/mp4", filename="result.mp4")

@app.post("/generate-from-audio")
async def generate_from_audio(image: UploadFile = File(...), audio: UploadFile = File(...)):
    image_path = save_upload_file(image, prefix="img")
    audio_path = save_upload_file(audio, prefix="aud")
    out_video = call_colab(image_path, audio_path)
    return FileResponse(out_video, media_type="video/mp4", filename="result.mp4")
```

Run backend:


Create `frontend/app.py`:

```python
from flask import Flask, render_template, request, redirect, url_for, send_file
import requests, os
from pathlib import Path

BACKEND_URL = "http://localhost:8000"  # FastAPI backend

app = Flask(__name__)
TMP_DIR = Path("tmp_frontend")
TMP_DIR.mkdir(exist_ok=True)

@app.route("/")
def index():
    return render_template("index.html")

@app.route("/submit_text", methods=["POST"])
def submit_text():
    img = request.files.get("image")
    text = request.form.get("text")
    if not img or not text:
        return "Provide image and text", 400

    files = {"image": (img.filename, img.stream, img.mimetype)}
    data = {"text": text}
    resp = requests.post(f"{BACKEND_URL}/generate-from-text", files=files, data=data, stream=True)
    if resp.status_code != 200:
        return f"Error: {resp.text}", 500

    out_path = TMP_DIR / "result_text.mp4"
    with open(out_path, "wb") as f:
        for chunk in resp.iter_content(chunk_size=1024*1024):
            if chunk:
                f.write(chunk)
    return redirect(url_for("show_video", filename="result_text.mp4"))

@app.route("/submit_audio", methods=["POST"])
def submit_audio():
    img = request.files.get("image")
    audio = request.files.get("audio")
    if not img or not audio:
        return "Provide image and audio", 400

    files = {
        "image": (img.filename, img.stream, img.mimetype),
        "audio": (audio.filename, audio.stream, audio.mimetype)
    }
    resp = requests.post(f"{BACKEND_URL}/generate-from-audio", files=files, stream=True)
    if resp.status_code != 200:
        return f"Error: {resp.text}", 500

    out_path = TMP_DIR / "result_audio.mp4"
    with open(out_path, "wb") as f:
        for chunk in resp.iter_content(chunk_size=1024*1024):
            if chunk:
                f.write(chunk)
    return redirect(url_for("show_video", filename="result_audio.mp4"))

@app.route("/video/<filename>")
def show_video(filename):
    path = TMP_DIR / filename
    if not path.exists():
        return "Video not found", 404
    return send_file(path, mimetype="video/mp4")

if __name__ == "__main__":
    app.run(debug=True, port=5000)
```

# 7 — Practical tips, gotchas & improvements

* **ngrok ephemeral URL**: each Colab session gives a new URL. For demo include instructions in README how to start Colab and update `colab_client.py`.
* **Runtime lifetime**: Colab disconnects after inactivity or ~90 minutes; keep it running during demo.
* **Model/checkpoint**: ensure the checkpoint is in `Wav2Lip/checkpoints/` in Colab.
* **Audio format**: convert TTS or uploaded audio to 16kHz mono WAV (pydub via `text_to_wav` already does).
* **Face quality**: Wav2Lip needs a clear frontal face. If image is low-res or angled, results degrade.
* **Large files**: enforce file size limits on backend (e.g., max 10 MB).
* **Timeouts**: Wav2Lip may take time—set `requests.post(..., timeout=300)` or longer.
* **Security**: sanitize and validate uploads. For assignment, ok to keep simple, but mention this in README.
* **Concurrency**: if multiple requests sent to Colab concurrently, performance may degrade; recommend queuing (for production: Celery + Redis + GPU pool).

# 9 — Suggested positive & negative findings content (copy into your submission)

**Positive findings**

* Wav2Lip produces convincing lip sync on frontal, well-lit faces.
* Using Colab as a GPU microservice is fast and cost-free for development/demo.
* Modular design (separate TTS, backend API, inference service) makes swapping TTS engines or hosting trivial.
* Using FastAPI + Flask kept the project fully Pythonic, easy to debug.

**Negative findings / limitations**

* Colab/ngrok are ephemeral; URL and runtime limits require manual restart during demo.
* Quality drops for angled/low-res faces; artifacts appear at mouth edges sometimes.
* gTTS prosody is limited — requires better TTS for natural intonation.
* CPU-only local runs of Wav2Lip are unacceptably slow, so GPU is essential.
* Concurrency: single Colab runtime is a bottleneck — production requires proper GPU service(s) and queueing.
